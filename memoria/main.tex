%    figuras/            imágenes (\includegraphics{nombre} las busca aquí)
%
%  Compilar: guarda en VS Code (LaTeX Workshop) o, desde la terminal,
%            latexmk -pdf main.tex      (pdflatex + biber)
%
%  No lleva portada: la añade la UPV al depositar el TFG (es la de los logos
%  de la UPV y la ETSINF que se ve en otros TFG).
%==============================================================================
% La clase carga tocbibind, que mete los índices como entradas del índice
% general. Se quita solo la del propio índice general; las de figuras y tablas
% se mantienen (con su página en números romanos).
\PassOptionsToPackage{nottoc}{tocbibind}
\documentclass[spanish,11pt,openany,listoffigures,listoftables]{tfgetsinf}

% Se anula la portada propia de la clase (ver arriba)
\makeatletter
\renewcommand{\m@ketitle}{}
\makeatother

% Sin páginas en blanco: cada capítulo empieza en la página siguiente, sea par
% o impar (openany), y \cleardoublepage, que la clase usa antes de los índices,
% pasa a ser un simple salto de página.
\let\cleardoublepage\clearpage
\addto\captionsspanish{\renewcommand{\titulacioname}{Grado en Ciencia de Datos}}

%------------------------------------------------------------------ Paquetes
% La clase ya carga babel, fontenc, mathpazo (Palatino), geometry, caption,
% listings, xcolor, graphicx, fancyhdr, titlesec, tocloft y hyperref.
% No los cargues otra vez (para los pies de figura usa \captionsetup).
\usepackage{amsmath,amssymb}
\usepackage{booktabs,multirow,array}
\usepackage{siunitx}
\sisetup{output-decimal-marker={,}}   % cifras con coma decimal
\usepackage{float}
\usepackage{subcaption}
\usepackage{enumitem}
\usepackage{microtype}
\graphicspath{{figuras/}{images/}}
\usepackage{threeparttable}

%------------------------------------------------------------------ Bibliografía
\usepackage{csquotes}                 % antes que biblatex
\usepackage[backend=biber,style=ieee,sorting=none]{biblatex}
\addbibresource{referencias.bib}
% El estilo IEEE no trae traducción al español de «Art. no.» (artículos con
% número de artículo en vez de páginas) y escribía «artno».

%------------------------------------------------------------------ Enlaces y referencias cruzadas
% Como en el TFG de Elena: referencias internas en negro, citas en verde,
% URL en cian y escritas con la misma letra que el texto.
\hypersetup{linkcolor=black,citecolor=green,urlcolor=cyan}
\urlstyle{same}

\usepackage[noabbrev,spanish]{cleveref}
% Sin esto cleveref escribe "la cuadro X" y "la apartado Y"

\crefname{table}{Tabla}{Tablas}               \Crefname{table}{Tabla}{Tablas}
\crefname{figure}{Figura}{Figuras}            \Crefname{figure}{Figura}{Figuras}
\crefname{equation}{Ecuación}{Ecuaciones}     \Crefname{equation}{Ecuación}{Ecuaciones}
\crefname{appendix}{Anexo}{Anexos}            \Crefname{appendix}{Anexo}{Anexos}

%------------------------------------------------------------------ Resúmenes
% Se redefine el entorno abstract de la clase para que el título sea un
% \chapter* (misma altura, letra, alineación y estilo de página que los
% índices y capítulos). Cada resumen empieza en página nueva. Tras el de
% inglés se ponen los índices, como hacía la clase.
\makeatletter
\renewenvironment{abstract}[1][catalan]{%
  \begingroup
    \selectlanguage{#1}%
    \chapter*{\abstractname}
    \@mkboth{\abstractname}{\abstractname}}{%
    \par\nopagebreak\noindent\textbf{\keywordsname:} \keywords\vspace*{3pt}\hrule
  \iflanguage{english}{\endgroup\makeindexes}{\endgroup\par}}
\makeatother
